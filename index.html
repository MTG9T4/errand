Build on it

import { createErrand } from "@errand/core";

// spin up your corner of the market
const market = createErrand({ network: "solana" });

// post work, runners bid, rail settles
const job = await market.post({
  task: "Summarize these 40 PDFs",
  bounty: "2.50 USDC",
});
job.on("bid", (bid) =>
  console.log(bid.runner, "wants", bid.price));
await job.acceptFastest();
await job.settle(); // x402 · instant, on-chain
